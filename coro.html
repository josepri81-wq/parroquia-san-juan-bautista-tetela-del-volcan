const narracion = document.getElementById("narracion"); const musica = document.getElementById("musica"); /* Volumen de la narración */ narracion.volume = 1.0; /* Música muy baja mientras habla la narración */ musica.volume = 0.08; /* Al comenzar la narración, bajar la música */ narracion.addEventListener("play", function () { musica.volume = 0.08; }); /* Al terminar la narración, recuperar la música */ narracion.addEventListener("ended", function () { musica.volume = 0.30; }); /* Si se pausa la narración, recuperar la música */ narracion.addEventListener("pause", function () { if (!narracion.ended) { musica.volume = 0.30; } }); /* Intentar reproducir ambos audios al entrar */ window.addEventListener("load", function () { musica.play().catch(function () { console.log("El navegador bloqueó temporalmente la música automática."); }); narracion.play().catch(function () { console.log("El navegador bloqueó temporalmente la narración automática."); }); });
